```python
async def generate(self, prompt: str, output_path: str) -> str:
    if not self.settings.seedream_api_key:
        raise SeedreamImageError("Seedream API key is not configured.")
    try:
        return await asyncio.wait_for(
            asyncio.to_thread(self._generate_sync, prompt, output_path),
            timeout=self.settings.seedream_timeout_seconds,
        )
    except TimeoutError as exc:
        raise SeedreamImageError(
            "Seedream master-image generation exceeded the five-minute limit."
        ) from exc

def _generate_sync(self, prompt: str, output_path: str) -> str:
    payload = {
        "model": self.settings.seedream_master_model,
        "prompt": prompt,
        "size": "2k",
        "response_format": "url",
        "watermark": False,
    }
    return self._post_and_download(payload, output_path)
```

3. 新增 helper 方法 _generate_master_seedream（放 _process_master 之后）：
```python
async def _generate_master_seedream(
    self, task_id: str, prompt_text: str, results_dir: str,
) -> str | None:
    if not self.settings.seedream_master_fallback_enabled:
        return None
    current = await self.task_service.get(task_id)
    if current.state in {TaskState.FAILED, TaskState.TIMEOUT}:
        await self.task_service.transition(task_id, TaskState.QUEUED)
    elif current.state != TaskState.QUEUED:
        logger.warning("Seedream fallback skipped for %s (state=%s)", task_id, current.state)
        return None
    await self.task_service.transition(task_id, TaskState.RUNNING)
    raw_dir = os.path.join(self.settings.output_dir, task_id, "seedream-raw")
    os.makedirs(raw_dir, exist_ok=True)
    raw_path = os.path.join(raw_dir, f"master-{uuid.uuid4().hex}.jpg")
    try:
        return await self.seedream.generate(prompt_text, raw_path)
    except SeedreamImageError as exc:
        logger.warning("Seedream master fallback failed for %s: %s", task_id, exc)
        await self._retry_or_finalize(task_id, TaskState.FAILED, str(exc)[:4000])
        return None
```
